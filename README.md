### Podman Compose

```yaml
services:
  blocky:
    image: "ghcr.io/daemonless/blocky:latest"
    container_name: blocky
    environment:
      - TZ=Etc/UTC  # Timezone for the container
    volumes:
      - "/path/to/containers/blocky:/config"
    ports:
      - "4000:4000"
      - "53:53"
      - "53:53"
    restart: unless-stopped
```
